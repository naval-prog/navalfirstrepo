function UserList() {
    const [users, setUsers] = useState([])
    useEffect(() => {
      fetch("https://api.example.com/users")
        .then(r => r.json())
        .then(data => setUsers(data))
    })
    return (
      <ul>
        {users.map(u => <li>{u.name}</li>)}
      </ul>
    )
}

1 mistake is  useEffect is without dependencies array it will render every update
Since setUsers() causes a re-render, this creates repeated API calls and may lead to an infinite loop.

2.this is correct version  is users.map((u) => <li key={u.id}>{u.name}</li>)
React requires a unique key prop when rendering lists so it can track and update items efficiently. Usually id is used as the key.
